cmake_minimum_required(VERSION 3.20.0)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(vns_rx LANGUAGES C)
if(NOT NCS_VERSION STREQUAL "3.3.0")
  message(FATAL_ERROR "NCS 3.3.0 is required; found '${NCS_VERSION}'")
endif()
if(NOT CONFIG_BOARD_NRF52DK_NRF52832)
  message(FATAL_ERROR "Use nrf52dk/nrf52832 with the RX app.overlay")
endif()
target_include_directories(app PRIVATE include)
target_compile_definitions(app PRIVATE VNS_ROLE_RX=1)
target_sources(app PRIVATE src/main.c src/rx_stim.c src/rx_hw.c src/rx_service.c src/rx_link_policy.c)
